class Cheerleader {
    #name;
    #height;

    constructor(name, height) {
        this.#name = name;
        this.#height = height;
    }

    getName() {
        return this.#name;
    }

    getHeight() {
        return this.#height;
    }
}

var stuntGroup = [];

function addCheerleader() {
    let name = document.getElementById("cheerleader").value;
    let height = Number(document.getElementById("height").value);

    stuntGroup.push(new Cheerleader(name, height));

    stuntGroup.sort(function(a, b) {
        return a.getHeight() - b.getHeight();
    });
}

function displayStuntGroup() {
    let namesColumn = "";

    for (let i = 0; i < stuntGroup.length; i++) {
        namesColumn += stuntGroup[i].getName() + " - "
            + stuntGroup[i].getHeight() + " cm<br>";
    }

    document.getElementById("stunt-group").innerHTML = namesColumn;
}
